use std::{env,fs,path::Path};
fn save(p:&Path,name:&str,x:&[f64]){fs::write(p.join(name),x.iter().flat_map(|v|v.to_le_bytes()).collect::<Vec<_>>()).unwrap();}
fn main(){
 let a:Vec<_>=env::args().collect();let mode=&a[1];let p=Path::new(&a[3]);fs::create_dir_all(p).unwrap();
 let x:Vec<f64>=fs::read(&a[2]).unwrap().chunks_exact(8).map(|b|f64::from_le_bytes(b.try_into().unwrap())).collect();
 let field=if a.get(4).is_some_and(|x|x=="diffuse"){iso532::FieldType::Diffuse}else{iso532::FieldType::Free};
 let nm:Vec<f64>=if mode=="nm"{x} else if mode=="core" {iso532::core::main_loudness::main_loudness(&x,field).unwrap().to_vec()} else {
  let(l,k)=iso532::zwtv::third_octave_levels::third_octave_levels_scalar(&x);
  let mut frames=vec![0.;28*k];for t in 0..k{for b in 0..28{frames[t*28+b]=l[b*k+t];}}
  let mut c=vec![0.;21*k];iso532::core::main_loudness::main_loudness_frames_into(&frames,k,field,&mut c).unwrap();
  let y=iso532::zwtv::nonlinear_decay::nl_loudness_scalar(&c,k);
  let mut row=vec![0.;21*k];for t in 0..k{for b in 0..21{row[t*21+b]=y[b*k+t];}}row
 };
 save(p,"nm.bin",&nm);
 let mut n24=vec![];let mut n22=vec![];let mut s24=vec![];let mut s22=vec![];
 for frame in nm.chunks_exact(21){let row:[f64;21]=frame.try_into().unwrap();let(n,s)=slopes24::calc_slopes(&row);n24.push(n);s24.extend(s);let(n,s)=slopes22::calc_slopes(&row);n22.push(n);s22.extend(s);}
 for(name,v)in[("n24.bin",&n24),("n22.bin",&n22),("s24.bin",&s24),("s22.bin",&s22)]{save(p,name,v);}
 if mode=="tv" {for(name,n)in[("weighted24.bin",&n24),("weighted22.bin",&n22)]{save(p,name,&iso532::zwtv::temporal_weighting::temporal_weighting(n));}}
}
