def array_check(x,shape):

            'shape_passed':x.shape==shape,'finite_count':int(np.isfinite(x).sum()),
            'nonfinite_count':int((~np.isfinite(x)).sum()),'passed':x.shape==shape and bool(np.isfinite(x).all())}
def compare(ref,got):
    """Same finite/shape/threshold comparator for primary and injected controls."""
    if ref.shape!=(241,) or got.shape!=(241,):
        return {'verdict':'NOT OBTAINABLE','reason':'shape','first_failure':None}
    if not np.isfinite(ref).all() or not np.isfinite(got).all():
        return {'verdict':'NOT OBTAINABLE','reason':'nonfinite','first_failure':None}
    d=np.abs(ref-got); bad=np.flatnonzero(d>1e-14)
    first=None if not len(bad) else ('total' if bad[0]==0 else f'specific[{bad[0]-1}]')
    verdict='AGREES' if not len(bad) else ('WITHIN TOLERANCE' if np.all(d<=np.maximum(.1,.05*np.abs(ref))) else 'DIFFERS')
    return {'verdict':verdict,'total_abs_diff':float(d[0]),'max_specific_abs_diff':float(d[1:].max()),
            'rounding_failure_count':len(bad),'first_failure':first,'differences':d.tolist()}
def invariance(a,b):
    assert a.shape==b.shape and np.isfinite(a).all() and np.isfinite(b).all()
    bad=np.argwhere(a.view(np.uint64)!=b.view(np.uint64))
    return {'compared_count':int(a.size),'bitwise_mismatch_count':len(bad),
            'max_abs_diff':float(np.abs(a-b).max()),
            'first_differing_coordinate':None if not len(bad) else bad[0].tolist(),
            'passed':not len(bad)}
def wav_read(p):
    with wave.open(str(p),'rb') as w:
        assert (w.getnchannels(),w.getsampwidth(),w.getframerate(),w.getcomptype())==(1,2,48000,'NONE')
        raw=w.readframes(w.getnframes());n=w.getnframes()
    decoded=np.frombuffer(raw,dtype='<i2').astype('<f8')/32768
    return decoded,raw,{'format':'PCM signed 16-bit little-endian','channels':1,'sample_rate':48000,
        'sample_width_bytes':2,'samples':n,'WAV_sha256':sha(p),'PCM_payload_sha256':hashlib.sha256(raw).hexdigest(),
        'decoded_f64le_sha256':ah(decoded),'file':p.name}
def wav_write(p,raw):
    with wave.open(str(p),'wb') as w:
        w.setnchannels(1);w.setsampwidth(2);w.setframerate(48000);w.writeframes(raw)

def run(cmd,cwd=None,check=True):
    t=datetime.datetime.now(datetime.timezone.utc).isoformat()
    p=subprocess.run(list(map(str,cmd)),cwd=cwd,capture_output=True,text=True,encoding='utf8',errors='replace')
    log.append({'started_utc':t,'command':list(map(str,cmd)),'cwd':str(cwd or repo),'exit_code':p.returncode,'stdout':p.stdout,'stderr':p.stderr})
    js(out/'run-log.json',log)
    if check and p.returncode: raise RuntimeError(p.stdout+p.stderr)
    return p
